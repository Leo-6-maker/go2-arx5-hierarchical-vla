def rpc(role,**args):

    result=call(role,RPC,ARGS=args,REGISTRY_SHA=__import__('hashlib').sha256((json.dumps(registry,indent=2,allow_nan=False)+'\n').encode()).hexdigest())
    if not result['ok']:
        raise RuntimeError(result.get('error_code','ENDPOINT_FAILURE_NO_RETRY'))
    return result.get('value')


def run_routes():
    sys.path[:0]=[str(REPO/'src'),str(REPO)]
    from scripts.pi0294.relay_transport import run_episode
    from scripts.pi0292.relay_protocol import unseal
    registry=json.loads((BASE/'MODEL_REGISTRY_02.private.json').read_text())
    directory=BASE/'model-M512-02-routes';directory.mkdir(mode=0o700,exist_ok=False)
    class Core:
        def observation(self,c,s):return rpc('core',operation='observation',episode=c['episode_uuid'],sequence=s)
        def teacher(self,c,s):raise PermissionError('NO_TEACHER_MODEL_INPUT')
        def install(self,c,s,b):return rpc('core',operation='install',episode=c['episode_uuid'],sequence=s,bundle=b)
        def ack(self,c,s):return rpc('core',operation='ack',episode=c['episode_uuid'],sequence=s)
    class Hopper:
        def query(self,c,q,teacher):
            assert teacher is None
            return rpc('hopper',operation='query',episode=c['episode_uuid'],query=q)
        def response(self,c,q):return rpc('hopper',operation='response',episode=c['episode_uuid'],query_uuid=q)
    records=[]
    for case in registry['cases'].values():
        folder=directory/case['episode_uuid'];folder.mkdir(mode=0o700)
        try:
            route=run_episode(folder,case,Core(),Hopper())
            records.append(dict(episode_uuid=case['episode_uuid'],attempt_id=case['attempt_id'],status=route['status']))
        except Exception as e:
            records.append(dict(episode_uuid=case['episode_uuid'],attempt_id=case['attempt_id'],status='STOPPED_PRESERVED_NO_RETRY',error_type=type(e).__name__,error_code=str(e)))
            try:rpc('core',operation='stop',episode=case['episode_uuid'])
            except Exception:pass
        save(directory/('PROGRESS-'+str(len(records)).zfill(2)+'.private.json'),dict(records=records))
    terminal=dict(status='SERIAL_LOCAL_MODEL_ROUTES_TERMINAL_PENDING_PHYSICAL_SCORING',records=records,model_calls_local=0,physics_integrations_local=0)
