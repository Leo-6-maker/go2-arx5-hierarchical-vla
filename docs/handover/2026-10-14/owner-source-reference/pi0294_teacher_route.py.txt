def call(host,code,**values):
    result=remote(host,'\n'.join(k+'='+repr(v) for k,v in values.items())+'\n'+code)
    if result.returncode:raise RuntimeError(result.stderr[-2000:])
    return json.loads(result.stdout)

def run():
    sys.path[:0]=[str(REPO/'src'),str(REPO)]
    from scripts.pi0294.relay_transport import run_episode
    from types import SimpleNamespace
    registry=json.loads((LOCAL/'REGISTRY.private.json').read_text())
    hashes={role:json.loads((LOCAL/(role+'-PREPARE.private.json')).read_text())['registry_sha256'] for role in ('core','hopper')}
    def rpc(role,**args):
        return call('corelab-4080s' if role=='core' else 'hopper',RPC,ROOT=CORE_ROOT if role=='core' else HOPPER_ROOT,
            ROLE=role,REGISTRY_SHA=hashes[role],ARGS=args)['value']
    def teacher(case,sequence):
        stop=time.time()+5
        while time.time()<stop:
            value=rpc('core',operation='teacher',episode=case['episode_uuid'],sequence=sequence)
            if value is not None:return value
            time.sleep(.05)
        raise TimeoutError('ACTUAL_FEEDBACK_TEACHER_NOT_AVAILABLE')
    core=SimpleNamespace(observation=lambda c,s:rpc('core',operation='observation',episode=c['episode_uuid'],sequence=s),
        teacher=teacher,
        install=lambda c,s,b:rpc('core',operation='install',episode=c['episode_uuid'],sequence=s,bundle=b),
        ack=lambda c,s:rpc('core',operation='ack',episode=c['episode_uuid'],sequence=s))
    hopper=SimpleNamespace(query=lambda c,q,t:rpc('hopper',operation='query',episode=c['episode_uuid'],query=q,teacher=t),
        response=lambda c,q:rpc('hopper',operation='response',episode=c['episode_uuid'],query_uuid=q))
    records=[]
    for case in registry['cases'].values():
        result=run_episode(LOCAL/case['episode_uuid'],case,core,hopper)
        records.append(result);save(LOCAL/('PROGRESS-'+str(len(records)).zfill(2)+'.private.json'),dict(records=records))
    terminal=dict(status='PASS_TWO_ACTUAL_TEACHER_ACK_CHAINS',records=records)
    # This is transport completion; the Core owner separately finishes audits.
    code="from pathlib import Path;import json;root=Path(ROOT);p=root/'evidence/teacher-relay-01/LOCAL_ROUTE_TERMINAL.json';assert not p.exists();p.write_text(json.dumps(VALUE));print(json.dumps({'status':VALUE['status']}))"
    call('corelab-4080s',code,ROOT=CORE_ROOT,VALUE=terminal)
    save(LOCAL/'TERMINAL.private.json',terminal)
