"""Read only authentication timestamps, never export credential values."""
import time
from pi0294_adopt import BASE, remote, save

CODE = r'''
import os,json,time,base64,re,datetime
from pathlib import Path
p=Path.home()/'.am/.amtoken'
raw=p.read_text()
match=re.search(r'[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+',raw)
v=dict(user=os.environ['USER'],home=str(Path.home()),mtime_epoch=p.stat().st_mtime,
       now_epoch=time.time(),credential_value_exported=False)
if match:
    try:
        payload=match.group().split('.')[1]
        claims=json.loads(base64.urlsafe_b64decode(payload+'='*((-len(payload))%4)))
        v['time_claims']={k:claims[k] for k in ('iat','exp','nbf')
                          if isinstance(claims.get(k),(int,float))}
        for k,x in list(v['time_claims'].items()):
            v['time_claims'][k+'_UTC']=datetime.datetime.fromtimestamp(x,datetime.timezone.utc).isoformat()
        if isinstance(claims.get('exp'),(int,float)):
            v['expired']=claims['exp']<time.time()
    except Exception as e:
        v['metadata_error_type']=type(e).__name__
else:
    v['jwt_detected']=False
print(json.dumps(v))
'''

if __name__ == '__main__':
    result=remote('hopper',CODE)
    save(BASE/('AUTH_DIAGNOSTIC_'+str(time.time_ns())+'.private.json'),
         dict(exit=result.returncode,stdout=result.stdout,stderr=result.stderr))
    print(result.stdout)
    if result.returncode:
        raise RuntimeError(result.stderr[-1000:])
