CUTOFF = 1791849137.0


def main(root):

    gate = json.loads((root/'PILOT_GATE.json').read_text())

    assert not any(row['id'].startswith(('T','D')) for row in initial['starts'])
    assert not any(row['status']=='STARTED' for row in initial['starts'])
    os.sched_setaffinity(0,set(sorted(os.sched_getaffinity(0))[:4]))
    env = dict(os.environ,PYTHONPATH=str(root/'code')+':'+str(root/'code/src'),
        MUJOCO_GL='egl',__EGL_VENDOR_LIBRARY_FILENAMES='/usr/share/glvnd/egl_vendor.d/50_mesa.json',
        LIBGL_ALWAYS_SOFTWARE='1',CUDA_VISIBLE_DEVICES='',LP_NUM_THREADS='4',OPENBLAS_NUM_THREADS='1')
    env.pop('MUJOCO_EGL_DEVICE_ID',None)
    python = '@CORELAB_NODE_STORAGE@/Projects/go2-arx5/PI0290/runtime/venv/bin/python'
    state = dict(owner='EXISTING_LOCAL_PC_ROOT_ONLY',pid=os.getpid(),source_sha=SOURCE,
        T0_epoch=1791830237,cutoff_epoch=CUTOFF,deadline_epoch=1791851837,
        phase='N3',order='TRAIN64_THEN_DEV16',completed_chunks=[],no_retries=True)
    path = root/'evidence/OWNER_N3_QUEUE_STATE.json'

        path.write_text(json.dumps(state,indent=2)+'\n')
        print(json.dumps(dict(owner_dispatch=state['current'],source_sha=SOURCE)),flush=True)
        result = subprocess.run([python,'-m','scripts.pi0291_arx5_data.run_queue',
            '--root',str(root),'--source-sha',SOURCE,'--phase',phase,'--start',str(start),'--count',str(count)],
            cwd=root/'code',env=env)
        state['completed_chunks'].append(dict(phase=phase,start=start,count=count,exit_code=result.returncode))
        state.update(current=None,updated_epoch=time.time())
        path.write_text(json.dumps(state,indent=2)+'\n')
        if result.returncode:
            state['stop']='CHILD_STOP_PRESERVE_FIRST_ATTEMPTS_NO_RETRY'
            break

            break
    else:

    state.update(current=None,updated_epoch=time.time(),physics_running=False)
    path.write_text(json.dumps(state,indent=2)+'\n')
