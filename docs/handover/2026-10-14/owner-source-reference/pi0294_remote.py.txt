def remote(host,code):
    if host not in ('corelab-amd-server','corelab-4080s') or not ROUTE.exists():
        return existing_remote(host,code)
    route=json.loads(ROUTE.read_text())
    assert route['owner']=='EXISTING_LOCAL_PC' and route['known_original_host_key_verified']
    assert route['actual_remote_hostname']=='corelab-amd-server' and route['actual_remote_uid']==2009
    assert route['StrictHostKeyChecking']=='yes' and route['BatchMode']=='yes'
    opts=['-o','BatchMode=yes','-o','ConnectTimeout=15','-o','ServerAliveInterval=10','-o','ServerAliveCountMax=3']
    command=shlex.join(['python3','-'])
    if host=='corelab-4080s':command=shlex.join(['ssh',*opts,host,command])
    args=['ssh',*opts,'-o','HostName='+route['hostname_override'],
        '-o','HostKeyAlias='+route['trusted_host_key_alias'],'-o','StrictHostKeyChecking=yes',
        route['ssh_config_alias'],command]
    return subprocess.run(args,input=code,capture_output=True,text=True,timeout=120)
