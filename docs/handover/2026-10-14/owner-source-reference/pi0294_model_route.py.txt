def run(step):
    sys.path[:0]=[str(REPO/'src'),str(REPO)]
    from scripts.pi0294.relay_transport import run_episode
    local=BASE/('model-M'+str(step));registry=json.loads((local/'REGISTRY.private.json').read_text())
    hashes={role:json.loads((local/(role+'-INSTALL.private.json')).read_text())['registry_sha256'] for role in ('core','hopper')}
    stage='model-M'+str(step)
    query_lock=Lock()
    def rpc(role,**args):
        # One resident inbox has a nonblocking writer lock; never retry a send.
        with query_lock if role=='hopper' and args['operation']=='query' else nullcontext():
            return call('corelab-4080s' if role=='core' else 'hopper',RPC,ROOT=CORE_ROOT if role=='core' else HOPPER_ROOT,
                ROLE=role,STAGE=stage,REGISTRY_SHA=hashes[role],ARGS=args)['value']
    core=SimpleNamespace(observation=lambda c,s:rpc('core',operation='observation',episode=c['episode_uuid'],sequence=s),
        teacher=lambda *args:(_ for _ in ()).throw(PermissionError('MODEL_CANNOT_READ_TEACHER')),
        install=lambda c,s,b:rpc('core',operation='install',episode=c['episode_uuid'],sequence=s,bundle=b),
        ack=lambda c,s:rpc('core',operation='ack',episode=c['episode_uuid'],sequence=s))
    hopper=SimpleNamespace(query=lambda c,q,t:rpc('hopper',operation='query',episode=c['episode_uuid'],query=q,teacher=None),
        response=lambda c,q:rpc('hopper',operation='response',episode=c['episode_uuid'],query_uuid=q))
    records=[]
    def one(case):
        try:return run_episode(local/case['episode_uuid'],case,core,hopper)
        except Exception:return json.loads((local/case['episode_uuid']/'ROUTE.json').read_text())
    with ThreadPoolExecutor(max_workers=4) as pool:
        futures=[pool.submit(one,c) for c in registry['cases'].values()]
        for future in as_completed(futures):
            records.append(future.result());save(local/('PROGRESS-'+str(len(records)).zfill(2)+'.private.json'),dict(records=records))
    value=dict(status='ALL_LOCAL_MODEL_ROUTE_OUTCOMES_PRESERVED_NOT_TASK_QUALIFICATION',records=records,
        model_calls_local=0,physics_integrations_local=0)
    save(local/'TERMINAL.private.json',value)
    code="from pathlib import Path;import json;root=Path(ROOT);p=root/'evidence'/STAGE/'LOCAL_ROUTE_TERMINAL.json';assert not p.exists();p.write_text(json.dumps(VALUE));print(json.dumps({'status':VALUE['status']}))"
    call('corelab-4080s',code,ROOT=CORE_ROOT,STAGE=stage,VALUE=value)
    print(json.dumps(dict(status=value['status'],episodes=len(records))))

        print(json.dumps(dict(status='INSTALLED_NOT_DISPATCHED',cases=len(registry['cases']))))
    elif args.action=='resident':
        assert args.sequence is not None
        registry_sha=json.loads((local/'hopper-INSTALL.private.json').read_text())['registry_sha256']
        value=call('hopper',RESIDENT,ROOT=HOPPER_ROOT,STEP=args.step,SEQUENCE=args.sequence,REGISTRY_SHA=registry_sha)
        save(local/'RESIDENT_DIRECTIVE.private.json',value);print(json.dumps(value))
    elif args.action=='core':
        hashes={role:json.loads((local/(role+'-INSTALL.private.json')).read_text())['registry_sha256'] for role in ('core','hopper')}
        proof=call('hopper',HOPPER_READY,ROOT=HOPPER_ROOT,STEP=args.step,REGISTRY_SHA=hashes['hopper'])
        save(local/'ACTUAL_RESIDENT_READY.private.json',proof)
        record=call('corelab-4080s',CORE_LAUNCH,ROOT=CORE_ROOT,STEP=args.step,REGISTRY_SHA=hashes['core'])
        save(local/'CORE_PROCESS.private.json',record);print(json.dumps(record))
    elif args.action=='closeout':
        value=json.loads((local/'TERMINAL.private.json').read_text())
        registry_sha=json.loads((local/'hopper-INSTALL.private.json').read_text())['registry_sha256']
        result=call('hopper',CLOSEOUT,ROOT=HOPPER_ROOT,STEP=args.step,VALUE=value,REGISTRY_SHA=registry_sha)
        save(local/'RESIDENT_STOP.private.json',result);print(json.dumps(result))
    else:run(args.step)
