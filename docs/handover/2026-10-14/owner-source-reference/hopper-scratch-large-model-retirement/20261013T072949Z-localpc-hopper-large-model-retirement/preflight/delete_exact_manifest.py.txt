def classify(st):
    if stat.S_ISREG(st.st_mode): return 'regular'
    if stat.S_ISDIR(st.st_mode): return 'directory'
    if stat.S_ISLNK(st.st_mode): return 'symlink'
    return 'special'

def matches(x):
    p=x['path']
    try: s=os.lstat(p)
    except FileNotFoundError: return False, 'missing'
    try: target=os.readlink(p) if stat.S_ISLNK(s.st_mode) else None
    except OSError as e: return False, f'readlink:{e}'
    want=(x.get('type'),x.get('uid'),x.get('gid'),x.get('dev'),x.get('ino'),x.get('nlink'),x.get('size_bytes'),x.get('blocks'),x.get('mode'),x.get('mtime_ns'),x.get('link_target'))
    got=(classify(s),s.st_uid,s.st_gid,s.st_dev,s.st_ino,s.st_nlink,s.st_size,s.st_blocks,stat.S_IMODE(s.st_mode),s.st_mtime_ns,target)
    return got == want, {'want':want,'got':got}
