import os,sys,json,stat
batch,root=sys.argv[1:3]
def rec(p):
 try:s=os.lstat(p)
 except FileNotFoundError:return {'event':'path','batch':batch,'root':root,'path':p,'kind':'member','exists':False}
 t='directory' if stat.S_ISDIR(s.st_mode) else 'regular' if stat.S_ISREG(s.st_mode) else 'symlink' if stat.S_ISLNK(s.st_mode) else 'special'
 x={'event':'path','batch':batch,'root':root,'path':p,'kind':'member','exists':True,'type':t,'uid':s.st_uid,'gid':s.st_gid,'dev':s.st_dev,'ino':s.st_ino,'nlink':s.st_nlink,'size_bytes':s.st_size,'blocks':s.st_blocks,'allocated_bytes':s.st_blocks*512,'mode':stat.S_IMODE(s.st_mode),'mtime_ns':s.st_mtime_ns}
 if t=='symlink':x['link_target']=os.readlink(p)
 return x
entries=[]
def walk(p):
 x=rec(p);entries.append(x)
 if not x.get('exists') or x.get('type')!='directory':return
 with os.scandir(p) as it: children=sorted([e.name for e in it])
 for name in children:walk(os.path.join(p,name))
walk(root)
