def command(args, timeout=12):
    try:
        r=subprocess.run(args,capture_output=True,text=True,timeout=timeout,env=dict(os.environ,GIT_OPTIONAL_LOCKS='0'))
        return dict(exit_code=r.returncode,stdout=r.stdout[:100000],stderr=r.stderr[:2000],truncated=len(r.stdout)>100000)
    except subprocess.TimeoutExpired:
        return dict(status='UNKNOWN_COMMAND_TIMEOUT')


def identity(p):
    row=dict(path=str(p),name=p.name)
    try:
        s=p.lstat()
        row.update(kind='symlink' if p.is_symlink() else 'directory' if p.is_dir() else 'file',
            realpath=str(p.resolve()),st_dev=s.st_dev,st_ino=s.st_ino,nlink=s.st_nlink,uid=s.st_uid,
            mode=oct(s.st_mode & 0o777),metadata_bytes=s.st_size,metadata_allocated_bytes=s.st_blocks*512,
            mtime_ns=s.st_mtime_ns,exists_following_link=p.exists())
        if p.is_symlink():
            row['link_text']=os.readlink(p)
            if p.exists():
                target=p.stat();row['target_identity']=dict(st_dev=target.st_dev,st_ino=target.st_ino)
    except OSError as e:
        row.update(status='UNKNOWN',error=type(e).__name__)
    return row


def git(p):
    def run(*args): return command(['git','-C',str(p),*args],timeout=20)
    result=dict(head=run('rev-parse','HEAD'),branch=run('symbolic-ref','--short','-q','HEAD'),
        git_dir=run('rev-parse','--absolute-git-dir'),common_dir=run('rev-parse','--git-common-dir'),
        worktrees=run('worktree','list','--porcelain'),status=run('status','--porcelain=v1','--untracked-files=normal'),
        local_branches=run('for-each-ref','--format=%(refname:short) %(objectname)','refs/heads'),
        origin_refs=run('for-each-ref','--format=%(refname:short) %(objectname)','refs/remotes/origin'),
        object_storage=run('count-objects','-v'))
    for name in ['AGENTS.md','pyproject.toml','setup.py','README.md']:
        q=p/name
        if name=='AGENTS.md' and q.is_file(): result['AGENTS_text']=q.read_text()[:16000]
    return result
