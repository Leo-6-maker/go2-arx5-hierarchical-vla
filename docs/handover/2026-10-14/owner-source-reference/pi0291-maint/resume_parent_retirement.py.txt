def minimal(pid):
    p=Path('/proc')/str(pid)
    if not p.exists():return dict(pid=pid,state='ABSENT')
    fields=(p/'stat').read_text().split(') ',1)[1].split()
    uid=next(x for x in (p/'status').read_text().splitlines() if x.startswith('Uid:')).split()[1:]
    return dict(pid=pid,state=fields[0],start_time_ticks=int(fields[19]),uid=uid)
