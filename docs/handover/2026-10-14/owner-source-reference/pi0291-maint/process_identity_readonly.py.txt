"""Print two direct /proc identity observations; never signal or write remotely."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import time

TARGETS = [3455361, 3455362]

def identity(pid):
    p = Path('/proc')/str(pid)
    try:
        stat = (p/'stat').read_text()
        tail = stat[stat.rfind(')')+2:].split()
        status = (p/'status').read_text()
        uid = next(s for s in status.splitlines() if s.startswith('Uid:')).split()[1:]
        fds = []
        for fd in sorted((p/'fd').iterdir(), key=lambda x:int(x.name)):
            try:
                target = os.readlink(fd)
                s = fd.stat()
                fds.append({'fd':int(fd.name), 'target':target, 'dev':s.st_dev,
                    'ino':s.st_ino,'nlink':s.st_nlink,'size':s.st_size,
                    'allocated_bytes':s.st_blocks*512})
            except OSError as e:
                fds.append({'fd':int(fd.name),'error':str(e)})
        return {'pid':pid,'state':tail[0],'ppid':int(tail[1]),'pgid':int(tail[2]),
            'sid':int(tail[3]),'start_time_ticks':int(tail[19]),'uid':uid,
            'exe':os.readlink(p/'exe'),'cwd':os.readlink(p/'cwd'),
            'cmdline':(p/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace'),
            'fds':fds, 'stat_sha256':hashlib.sha256(stat.encode()).hexdigest()}
    except OSError as e:
        return {'pid':pid,'error':str(e)}

rows=[]
for i in range(2):
    own=[]
    all_pids=[]
    for p in Path('/proc').iterdir():
        if not p.name.isdigit():continue
        try:
            if p.stat().st_uid != os.getuid():continue
            s=(p/'stat').read_text();tail=s[s.rfind(')')+2:].split()
            cmd=(p/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace')
            all_pids.append({'pid':int(p.name),'ppid':int(tail[1]),'state':tail[0],'cmdline':cmd})
        except OSError:continue
    descendants=[]; parents=set(TARGETS)
    while True:
        add=[x for x in all_pids if x['ppid'] in parents and x['pid'] not in parents]
        if not add:break
        descendants+=add;parents.update(x['pid'] for x in add)
    rows.append({'observation':i+1, 'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
        'hostname':os.uname().nodename,'probe_uid':os.getuid(),
        'targets':[identity(p) for p in TARGETS],'descendants':descendants,
        'current_science_role_processes':[x for x in all_pids
            if any(z in x['cmdline'] for z in ['pi0290_task_data/run', 'pi0290_task_data.driver',
                'PI0290/code/driver', 'PI0290/code/run_episode', 'PI0290/code/runner'])]})
    if i==0:time.sleep(2)
print(json.dumps({'schema':'PRIVATE_TWO_OBSERVATION_PROC_IDENTITY',
    'observations':rows,'no_signal':True,'no_remote_mutation':True},indent=2))
