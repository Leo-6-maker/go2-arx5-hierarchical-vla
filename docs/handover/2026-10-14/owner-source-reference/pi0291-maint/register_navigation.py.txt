"""Private bounded host-aware index: stat only, never relocate existing assets."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import socket
import time

parser = argparse.ArgumentParser()
parser.add_argument('--node', choices=['corelab-amd-server', 'corelab-4080s'], required=True)
parser.add_argument('--migration', required=True)
parser.add_argument('--output', required=True)
args = parser.parse_args()
began = time.process_time()
shared = Path('@LOCAL_OR_CORE_HOME@-nfs/Projects/go2-arx5')
ssd = Path('@CORELAB_NODE_STORAGE@/Projects/go2-arx5')
run = ssd / 'PI0291/20261013T023717-localpc-arx5'
rows = []


def add(role, path, consumer, lifecycle):
    path = Path(path)
    row = dict(role=role, logical_path=str(path), consumer=consumer,
               lifecycle=lifecycle, observed_on=args.node)
    if consumer == 'corelab-4080s' and args.node != consumer:
        row.update(status='NODE_LOCAL_QUERY_EXECUTOR_BY_SSH', local_absence_is_not_loss=True)
    else:
        try:
            st = path.stat()
            row.update(status='OBSERVED', realpath=str(path.resolve()), st_dev=st.st_dev,
                       st_ino=st.st_ino, nlink=st.st_nlink, uid=st.st_uid,
                       bytes=st.st_size, allocated=st.st_blocks*512)
        except OSError as exc:
            row.update(status='UNKNOWN', error=f'{type(exc).__name__}: {exc}')
    rows.append(row)


add('SHARED_NFS_OWNER_ROOT', shared, 'BOTH_CORELAB_NODES', 'KEEP')
add('HOME_SHARED_ORIGINALS', '@LOCAL_OR_CORE_HOME@-nfs', 'BOTH_CORELAB_NODES', 'KEEP')
add('4080S_SSD_ROOT', ssd, 'corelab-4080s', 'KEEP')
add('PI0290_PROTECTED_RUNTIME', ssd/'PI0290/runtime/venv/bin/python', 'corelab-4080s', 'PROTECT')
add('PI0290_PROTECTED_ORIGINALS', ssd/'PI0290', 'corelab-4080s', 'PROTECT')
for role, path in [('WORK', run), ('RAW', run/'episodes'), ('EVIDENCE', run/'evidence'),
                   ('SOURCE', run/'code'), ('PATH_BINDING', run/'PRIVATE_PATH_INDEX.json')]:
    add('PI0291_'+role, path, 'corelab-4080s', 'ACTIVE_OWNER_EXCLUDED_FROM_MAINTENANCE')
add('PRIVATE_WBC_NATIVE_SOURCE', shared/'private/wbc_mujoco_minimal_20261009_6dd13098',
    'BOTH_CORELAB_NODES', 'READ_ONLY_PROTECTED')
migration = json.loads(Path(args.migration).read_text())
for card in migration['candidates']:
    add('LEGACY_SOURCE_RETAINED', card['source'], 'BOTH_CORELAB_NODES', 'KEEP')
    if card['status'] == 'COPY_READBACK_VERIFIED_SOURCE_RETAINED':
        add('VERIFIED_NFS_COPY', card['destination'], 'BOTH_CORELAB_NODES', 'COPY_NOT_SPACE_RECLAMATION')
value = dict(schema='PRIVATE_HOST_AWARE_STORAGE_NAVIGATION_V1', node=args.node,
             actual_hostname=socket.gethostname(), epoch=time.time(), entries=rows,
             modified_existing_paths=0, file_deletions=0, symlink_rewrites=0,
             quota='UNKNOWN', CPU_s=time.process_time()-began)
out = Path(args.output)
if out.exists():
    raise FileExistsError('Preserve existing navigation observation')
out.write_text(json.dumps(value, indent=2, allow_nan=False)+'\n')
os.chmod(out, 0o600)
assert json.loads(out.read_text()) == value
print(json.dumps(dict(node=args.node, entries=len(rows),
    observed=sum(row['status']=='OBSERVED' for row in rows),
    host_local_deferred=sum(row['status']=='NODE_LOCAL_QUERY_EXECUTOR_BY_SSH' for row in rows),
    unknown=sum(row['status']=='UNKNOWN' for row in rows), CPU_s=value['CPU_s'],
    sha256=hashlib.sha256(out.read_bytes()).hexdigest(), readback=True)))
