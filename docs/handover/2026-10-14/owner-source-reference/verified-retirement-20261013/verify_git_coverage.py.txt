def sha(p):

def walk(p,exclude_git=False):

  if s.st_dev!=dev or f!=p and f.resolve() in mounts:raise RuntimeError('nested mount '+str(f))
  if s.st_uid!=os.getuid():raise RuntimeError('not owned '+str(f))
  if stat.S_ISLNK(s.st_mode):r=dict(relative=rel,type='symlink',target=os.readlink(f),identity=i)
  elif stat.S_ISREG(s.st_mode):
   h=sha(f);after=ident(f)
   if i!=after:raise RuntimeError('file changed while hashing '+str(f))
   r=dict(relative=rel,type='file',sha256=h,identity=i)
  elif stat.S_ISDIR(s.st_mode):
   r=dict(relative=rel,type='directory',identity=i)
   stack.extend(reversed(sorted(x for x in f.iterdir() if not(exclude_git and x.name=='.git'))))
  else:raise RuntimeError('special file '+str(f))
