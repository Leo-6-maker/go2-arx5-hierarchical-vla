def watch():

        value=remote('hopper',STATUS,label='queue-wait-readback')
        job=value.get('job',{});state=job.get('job_state')
        if state=='R' and value.get('pbs/alloc-01/PBS_READY.json'):
            print(json.dumps(dict(status='NATIVE_RUNNING_OWNER_READY',job_id=value['job_id'],native_start=job.get('stime'))),flush=True)
            return remote('hopper',RELEASE_CODE,label='PBS-release')
        if state in ['F','X']:
            return dict(status='NATIVE_TERMINAL_BEFORE_RELEASE',job_id=value['job_id'],exit=job.get('Exit_status'),final=value.get('pbs/alloc-01/PBS_PROCESS_FINAL.json'))
        time.sleep(5 if state=='R' else 20)
    return dict(status='E0_QUEUE_WINDOW_ENDED_REQUIRES_EXACT_OWN_QUEUE_WITHDRAWAL')
