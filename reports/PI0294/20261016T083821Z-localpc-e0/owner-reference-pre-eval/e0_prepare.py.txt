def encoded(value):
    return json.dumps(value, indent=2, allow_nan=False)+'\n'


def digest(value):
    return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()


def snapshot(label, role):
    paths = sorted(BASE.glob(f'{label}-{role}-*.private.json'))
    return json.loads(json.loads(paths[-1].read_text())['stdout'])


def package():
    a = adopt()
    PUBLIC.mkdir(parents=True, exist_ok=True)
    for name, item in a['reused_preparation_files'].items():
        data = (PREP/name).read_bytes()
        assert hashlib.sha256(data).hexdigest() == item['sha256']
        destination = PUBLIC/'preparation'/name
        destination.parent.mkdir(exist_ok=True)
        if destination.exists():
            assert destination.read_bytes() == data
        else:
            destination.write_bytes(data)
    public = {k:v for k,v in a.items() if k != 'exact_private_roots'}
    public.update(status='ADOPTED_VALID_NEW_E0_WINDOW_BINDING_AND_RUNTIME_PREPARATION',

    (PUBLIC/'E0_ADOPTION_AND_BINDING.json').write_text(encoded(public))
    old = snapshot('original-panel','core')['evidence/model-M2048/REGISTRY.json']
    assert len(old['cases']) == 12
    conditions, private_cases = [], []
    for n,c in enumerate(sorted(old['cases'].values(),key=lambda x:(x['card']['split']!='TRAIN',x['layout_id'],x['card']['task']))):
        original = {k:v for k,v in c['card'].items() if not k.startswith('relay_request_')}

        row = dict(index=n,attempt_id=card['id'],original_attempt_id=original['id'],
            split=card['split'],task=card['task'],scene_group=card['scene_group'],
            initial_state_sha256=card['initial_state_sha256'],original_card_sha256=digest(original),
            execution_card_sha256=digest(card),card=card,
            randomness_rule='sha256(PI0294/<original_scene_group>/<prefix_sequence>)[:8]_big_endian_mod_2**63',
            prefix0_seed=int.from_bytes(hashlib.sha256(('PI0294/'+card['scene_group']+'/0').encode()).digest()[:8],'big')%(2**63),
            status='NOT_RUN')
        conditions.append(row)
        private_cases.append(dict(row,episode_uuid=str(uuid.uuid4())))
    panel = dict(schema='PI0294_E0_FIXED_ORIGINAL_M2048_PANEL_V1',run_uuid=RUN,protocol_ref=RELEASE,

    assert sum(c['split']=='TRAIN' for c in conditions) == 4
    assert sum(c['split']=='DEV' for c in conditions) == 8
    (PUBLIC/'E0_PANEL_MANIFEST.json').write_text(encoded(panel))
    if not (BASE/'CASES.private.json').exists():
        save(BASE/'CASES.private.json',private_cases)
    else:
        previous=json.loads((BASE/'CASES.private.json').read_text())
        assert [{k:v for k,v in c.items() if k!='episode_uuid'} for c in previous]==conditions
    report = dict(status='ACTUAL_READ_ONLY_BINDING_PREFLIGHT_NO_MODEL_OR_PHYSICS',
        authentication='EXPLICIT_USER_AUTHORIZED_LOGIN_THEN_NATIVE_PROJECT_QUERY_SUCCEEDED',
        project='CFP04-CF-007',project_active_and_member=True,own_Hopper_jobs_observed=0,
        personal_scratch_used_bytes=426893307904,personal_scratch_hard_bytes=1099511627776,
        original_M2048_training_folder='METADATA_PRESENT_WEIGHT_SHARDS_MISSING',
        missing_inference_member_action='CHECK_REGISTERED_COPIES_THEN_FIXED_PUBLIC_HF_ONLY_NO_OPTIMIZER',
        source_D0_engine_scene_episode_scorer_S1_numeric_bytes_match_qualification=True,
        fullbody_TF_R3='NATIVE_SAFETY_STOP_PRESERVED_OWNER_LOCKS_UNHELD_NO_RESTART',
        no_new_PBS_model_queries_training_or_physics=True,
        private_snapshots_retained=True,credentials_and_user_roster_published=False)
    (PUBLIC/'PREFLIGHT_ACTUAL.json').write_text(encoded(report))
    return dict(status='PUBLIC_SANITIZED_BINDING_AND_12_ORIGINAL_CONDITIONS_PREPARED',files=8,run_uuid=RUN)

def install(role):
    source=subprocess.check_output(['git','rev-parse','HEAD'],cwd=REPO,text=True).strip()
    paths=subprocess.check_output(['git','ls-tree','-r','--name-only',source,'src','scripts/pi0291_arx5_data',
        'scripts/pi0292','scripts/pi0294','scripts/iteration'],cwd=REPO,text=True).splitlines()
    contents={n:base64.b64encode(subprocess.check_output(['git','show',source+':'+n],cwd=REPO)).decode() for n in paths}
    numeric_names=['scripts/pi0294/episode.py','scripts/pi0294/scene.py','scripts/pi0294/scene_cards.py',
        'scripts/pi0294/teacher.py','scripts/pi0294/score.py','scripts/pi0294/s1_preflight.py',
        'scripts/pi0292/physics_relay.py','scripts/pi0292/relay_protocol.py','scripts/pi0292/live_inputs.py',
        'scripts/pi0292/eval_contract.py','scripts/pi0292/train_adapter.py','scripts/pi0291_arx5_data/config.py',
        'scripts/pi0291_arx5_data/engine.py','scripts/pi0294/numerics.py','scripts/pi0294/local_backbone.py']
    numeric={}
    for name in numeric_names:
        original=subprocess.check_output(['git','show','90c90d784dd702f10c79c719d717e80aeb50e31e:'+name],cwd=REPO)
        assert original==base64.b64decode(contents[name]),name
        numeric[name]=hashlib.sha256(original).hexdigest()
    parent=HOPPER if role=='hopper' else CORE
    inventory=json.loads((REPO/'reports/PI0294'/R3/'HF_PUBLIC_M2048_RECEIPT.json').read_text())
    names=['config.json','model.safetensors.index.json','statistics.json','processor/embodiment_id.json',
        'processor/processor_config.json','processor/statistics.json']+[f'model-{i:05d}-of-00003.safetensors' for i in range(1,4)]
    values=dict(ROOT=parent+'/'+RUN,OLD=parent+'/'+R3,RUN=RUN,R3=R3,RELEASE=RELEASE,SOURCE=source,
        ROLE='PI0294_E0_HOPPER_RUN' if role=='hopper' else 'PI0294_E0_4080_RUN',HOPPER_ROLE=role=='hopper',
        LIMITS=e0_scope.LIMITS,STATS=e0_scope.STATS,CONTENTS=contents,NUMERIC=numeric,
        INVENTORY=inventory,INFERENCE_NAMES=names)
    return call(role,'\n'.join(k+'='+repr(v) for k,v in values.items())+'\n'+INSTALL,label='source-staging')
