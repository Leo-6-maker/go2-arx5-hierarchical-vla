def audits():
    progress=json.loads((BASE/'E0_RESULTS_PROGRESS.private.json').read_text())
    registry=json.loads((BASE/'REGISTRY.private.json').read_text())
    byid={c['attempt_id']:c for c in registry['cases'].values()}
    rows=[]
    for row in progress['rows']:
        case=byid[row['attempt_id']]
        audit_start(case)
        value={}
        end=time.time()+650
        while time.time()<end:
            value=audit_read(case)
            if value['terminal']:break
            time.sleep(2)
        save(BASE/('AUDIT-'+case['attempt_id']+'.private.json'),value)
        assert value.get('terminal') and value.get('audit'), 'AUDIT_TERMINAL_MISSING_INSPECT_LOG_NO_REPLAY'
        actual=value['audit'];rows.append(actual)
        save(BASE/'E0_AUDITS_PROGRESS.private.json',dict(rows=rows),exclusive=False)
        print(json.dumps(dict(attempt=case['attempt_id'],status=actual['status'],errors=actual['errors'],video_frames=(actual.get('video') or {}).get('actual_paired_frames'))),flush=True)
    return dict(audited=len(rows),pass_count=sum(r['status']=='PASS_INDEPENDENT_SAVED_EVIDENCE_AUDIT' for r in rows))
