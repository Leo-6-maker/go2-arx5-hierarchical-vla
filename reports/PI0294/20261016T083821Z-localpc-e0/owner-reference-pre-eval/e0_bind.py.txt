def remote(role, body, values=None, *, label):
    parent=HOPPER if role=='hopper' else CORE
    args=dict(ROOT=parent+'/'+RUN,RUN=RUN)
    args.update(values or {})
    return call(role,'\n'.join(k+'='+repr(v) for k,v in args.items())+'\n'+body,label=label)
