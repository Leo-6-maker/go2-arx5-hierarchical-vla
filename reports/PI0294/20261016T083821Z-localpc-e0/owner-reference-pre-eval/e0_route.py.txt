def rpc(role, **args):
    registry=json.loads((BASE/'REGISTRY.private.json').read_text())
    digest=hashlib.sha256(encoded(registry).encode()).hexdigest()
    result=remote(role,RPC,dict(ROLE=role,ARGS=args,REGISTRY_SHA=digest),label='RPC-'+args['operation'])
    if not result['ok']:
        if result.get('known_terminal'):
            raise RuntimeError('KNOWN_CORE_TERMINAL:'+result['known_terminal']['failure_stage'])
        raise RuntimeError('ONE_SHOT_ENDPOINT_OUTCOME_UNKNOWN_NO_RETRY')
    return result['value']

def routes():
    ready=readiness();registry=json.loads((BASE/'REGISTRY.private.json').read_text())
    ordered=sorted(registry['cases'].values(),key=lambda c:c['panel_index'])
    route_root=BASE/'routes';route_root.mkdir(exist_ok=True,mode=0o700)
    class CoreEndpoint:
        def observation(self,c,s):return rpc('core',operation='observation',episode=c['episode_uuid'],sequence=s)
        def ack(self,c,s):return rpc('core',operation='ack',episode=c['episode_uuid'],sequence=s)
        def teacher(self,c,s):raise PermissionError('NO_TEACHER_LABELS_TO_E0_MODEL')
        def install(self,c,s,b):return rpc('core',operation='install',episode=c['episode_uuid'],sequence=s,bundle=b)
    class HopperEndpoint:
        last=None
        def query(self,c,q,teacher):
            assert teacher is None
            self.last=q
            return rpc('hopper',operation='query',episode=c['episode_uuid'],query=q)
        def response(self,c,q):return rpc('hopper',operation='response',episode=c['episode_uuid'],query_uuid=q)
    core,hopper=CoreEndpoint(),HopperEndpoint();rows=[]
    for i,case in enumerate(ordered):
        if time.time()+900>=ready['allocation_end_epoch']-1800:break
        hopper.last=None
        if i==4:
            save(BASE/'E0_TRAIN4_INTERIM.private.json',dict(rows=rows,no_task_success_gate=True,continue_DEV8=all(r['pipeline_trustworthy'] for r in rows)))
        start(case)
        failure=None
        try:run_episode(route_root/case['episode_uuid'],case,core,hopper)
        except BaseException as error:
            failure=dict(type=type(error).__name__,message=str(error))
            rejection=None
            if hopper.last:
                query=unseal(hopper.last)
                try:
                    action=hopper.response(case,query['query_uuid'])
                    if action:
                        physical=unseal(action)['payload']['trajectory_eef10']
                        gate=gate_action(physical,query['payload']['state_eef10'],query['tick']*.002,query['reference_uuid'])
                        if gate['decision']=='REJECT':rejection=dict(original_gate=gate,action_sha256=action['sha256'],query_uuid=query['query_uuid'],actual_returned_action_preserved=True,no_resampling=True)
                except Exception:pass
            stop_case(case,rejection)
        end=time.time()+90;value={}
        while time.time()<end:
            value=terminal(case)
            if value['worker_terminal']:break
            time.sleep(.5)
        save(BASE/('CASE-'+case['attempt_id']+'.private.json'),dict(terminal=value,route_failure=failure))
        if 'E0_ADJUDICATION.json' not in value:
            raise RuntimeError('E0_WORKER_ENGINEERING_TERMINAL_WITHOUT_ADJUDICATION')
        row=value['E0_ADJUDICATION.json'];rows.append(row)
        save(BASE/'E0_RESULTS_PROGRESS.private.json',dict(rows=rows,covered=len(rows),planned=12),exclusive=False)
        print(json.dumps(dict(covered=len(rows),planned=12,attempt=row['attempt_id'],outcome=row['status'],stage=row['failure_stage'],steps=row['main_integrations'],pipeline_trustworthy=row['pipeline_trustworthy'])),flush=True)
        if not row['pipeline_trustworthy']:break
    return dict(rows=rows,covered=len(rows),planned=12,training_updates=0)
